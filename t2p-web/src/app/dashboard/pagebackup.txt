'use client';

import { useState } from 'react';
import axios from 'axios';
import Link from 'next/link';
import {
  RadarChart, PolarGrid, PolarAngleAxis, PolarRadiusAxis,
  Radar, BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, Legend
} from 'recharts';
import { Loader2, ArrowLeft, Search } from 'lucide-react';

const API_BASE = process.env.NEXT_PUBLIC_API_URL || 'http://127.0.0.1:8001';

// 辅助：截断长主题名（用于图表显示）
const truncateTopic = (name: string, maxLen = 11) =>
  name.length > maxLen ? name.slice(0, maxLen) + '…' : name;

export default function Dashboard() {
  const [userId, setUserId] = useState('');
  const [data, setData] = useState<any>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  const fetchProgress = async () => {
    if (!userId.trim()) return;
    setLoading(true);
    setError('');
    try {
      const res = await axios.get(`${API_BASE}/user/progress/${userId.trim()}`);
      if (res.data.success) {
        setData(res.data.topics_status);
      } else {
        setError('未能获取数据，请检查用户名。');
      }
    } catch (e) {
      setError('获取数据失败，请确认后端已启动。');
    } finally {
      setLoading(false);
    }
  };

  // 雷达图数据（熟练度百分比）
  const radarData = data
    ? Object.entries(data).map(([topic, info]: any) => ({
        topic: truncateTopic(topic),
        mastery: Math.round(info.mastery_score * 100),
      }))
    : [];

  // 柱状图数据（等级 + 练习次数）
  const barData = data
    ? Object.entries(data).map(([topic, info]: any) => ({
        topic: truncateTopic(topic),
        level: info.current_level,
        count: info.completed_count,
      }))
    : [];

  return (
    <div className="min-h-screen bg-[#FBFBFE] pb-20">
      <div className="h-1.5 bg-gradient-to-r from-indigo-600 to-violet-600" />
      <main className="max-w-4xl mx-auto py-12 px-6">
        {/* 顶部导航 */}
        <div className="flex items-center justify-between mb-10">
          <Link
            href="/"
            className="flex items-center gap-2 text-slate-500 hover:text-indigo-600 text-xs font-bold uppercase tracking-wider transition-colors"
          >
            <ArrowLeft size={16} />
            返回练习
          </Link>
        </div>

        {/* 查询区域 */}
        <div className="bg-white rounded-[2.5rem] p-8 shadow-sm border border-slate-200/50 mb-10">
          <div className="flex items-end gap-4">
            <div className="flex-1">
              <label className="text-[10px] font-black text-slate-400 uppercase tracking-widest mb-2 block">
                请输入用户名
              </label>
              <input
                type="text"
                value={userId}
                onChange={(e) => setUserId(e.target.value)}
                onKeyDown={(e) => e.key === 'Enter' && fetchProgress()}
                placeholder="输入已登录的用户名"
                className="w-full bg-slate-50 border-none rounded-2xl px-4 py-3.5 font-bold text-slate-700 focus:ring-2 focus:ring-indigo-500 outline-none"
              />
            </div>
            <button
              onClick={fetchProgress}
              disabled={loading || !userId.trim()}
              className="flex items-center justify-center gap-2 bg-slate-900 text-white px-8 py-3.5 rounded-2xl font-bold text-xs hover:bg-indigo-600 transition-all shadow-lg disabled:opacity-50"
            >
              {loading ? <Loader2 size={16} className="animate-spin" /> : <Search size={16} />}
              查看看板
            </button>
          </div>
          {error && <p className="mt-4 text-red-500 text-xs font-bold">{error}</p>}
        </div>

        {/* 图表区 */}
        {data && (
          <div className="space-y-8">
            <div className="bg-white rounded-[3rem] p-8 shadow-sm border border-slate-200/50">
              <h2 className="text-sm font-bold text-slate-700 mb-6">📐 主题熟练度雷达图</h2>
              <div className="flex justify-center">
                <RadarChart width={500} height={400} data={radarData}>
                  <PolarGrid stroke="#e2e8f0" />
                  <PolarAngleAxis dataKey="topic" tick={{ fontSize: 11, fill: '#64748b' }} />
                  <PolarRadiusAxis angle={30} domain={[0, 100]} tick={{ fontSize: 10 }} />
                  <Radar
                    name="熟练度 (%)"
                    dataKey="mastery"
                    stroke="#6366f1"
                    fill="#6366f1"
                    fillOpacity={0.3}
                  />
                </RadarChart>
              </div>
            </div>

            <div className="bg-white rounded-[3rem] p-8 shadow-sm border border-slate-200/50">
              <h2 className="text-sm font-bold text-slate-700 mb-6">📊 等级与练习次数</h2>
              <div className="flex justify-center">
                <BarChart width={500} height={400} data={barData}>
                  <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
                  <XAxis dataKey="topic" tick={{ fontSize: 11, fill: '#64748b' }} />
                  <YAxis />
                  <Tooltip />
                  <Legend />
                  <Bar dataKey="level" fill="#10b981" name="当前等级" radius={[4, 4, 0, 0]} />
                  <Bar dataKey="count" fill="#f59e0b" name="练习次数" radius={[4, 4, 0, 0]} />
                </BarChart>
              </div>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
